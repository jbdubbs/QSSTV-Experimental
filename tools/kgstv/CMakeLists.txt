cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(qsstv_kgstv LANGUAGES CXX)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui)
    include(CTest)
endif()
set(CMAKE_AUTOMOC ON)
set(KGSTV_DIR "${CMAKE_CURRENT_LIST_DIR}/../../src/kgstv")
add_library(kgstv_core STATIC
    ${KGSTV_DIR}/kgstvprotocol.cpp ${KGSTV_DIR}/kgstvaudio.cpp
    ${KGSTV_DIR}/kgstvimage.cpp ${KGSTV_DIR}/kgstvtransfer.cpp
    ${KGSTV_DIR}/kgstvcontroller.cpp ${KGSTV_DIR}/kgstvcontroller.h)
target_include_directories(kgstv_core PUBLIC "${KGSTV_DIR}")
target_link_libraries(kgstv_core PUBLIC Qt6::Core Qt6::Gui)
target_compile_features(kgstv_core PUBLIC cxx_std_17)
add_executable(kgstv-offline offline.cpp)
target_link_libraries(kgstv-offline PRIVATE kgstv_core)
if(BUILD_TESTING)
    foreach(kind core controller reference)
        add_executable(kgstv_test_${kind} ../../tests/kgstv/test_${kind}.cpp)
        target_link_libraries(kgstv_test_${kind} PRIVATE kgstv_core)
    endforeach()
    add_test(NAME kgstv_core COMMAND kgstv_test_core)
    add_test(NAME kgstv_controller COMMAND kgstv_test_controller)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
        add_test(NAME kgstv_offline COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_LIST_DIR}/../../tests/kgstv/test_offline.py $<TARGET_FILE:kgstv-offline>)
    endif()
endif()
